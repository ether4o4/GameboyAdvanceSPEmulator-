cmake_minimum_required(VERSION 3.22.1)
project(watchboy_native C CXX)

set(LIBMGBA_ONLY ON CACHE BOOL "Build only the mGBA core" FORCE)
set(DISABLE_DEPS ON CACHE BOOL "Use mGBA's bundled core dependencies" FORCE)
set(DISABLE_FRONTENDS ON CACHE BOOL "Omit desktop frontends" FORCE)
set(SKIP_LIBRARY ON CACHE BOOL "Build the libretro core only" FORCE)
set(BUILD_LIBRETRO ON CACHE BOOL "Build the mGBA libretro core" FORCE)
set(M_CORE_GBA ON CACHE BOOL "Build GBA emulation" FORCE)
set(M_CORE_GB OFF CACHE BOOL "Omit Game Boy emulation" FORCE)
set(BUILD_QT OFF CACHE BOOL "Omit Qt" FORCE)
set(BUILD_SDL OFF CACHE BOOL "Omit SDL" FORCE)
set(BUILD_GL OFF CACHE BOOL "Omit desktop OpenGL" FORCE)
set(BUILD_GLES2 OFF CACHE BOOL "Omit GLES frontend" FORCE)
set(BUILD_GLES3 OFF CACHE BOOL "Omit GLES frontend" FORCE)
set(BUILD_TEST OFF CACHE BOOL "Omit upstream tests" FORCE)
set(BUILD_SUITE OFF CACHE BOOL "Omit upstream test suite" FORCE)
set(BUILD_PERF OFF CACHE BOOL "Omit desktop performance tool" FORCE)
set(ENABLE_DEBUGGERS OFF CACHE BOOL "Omit debugger support" FORCE)
set(ENABLE_SCRIPTING OFF CACHE BOOL "Omit scripting" FORCE)

add_subdirectory(third_party/mgba mgba-core)
set_target_properties(mgba_libretro PROPERTIES PREFIX "lib")

add_library(watchboy SHARED watchboy_jni.c)
target_include_directories(watchboy PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/mgba/src/platform/libretro")
target_compile_options(watchboy PRIVATE -Wall -Wextra -Werror=return-type)
target_link_libraries(watchboy PRIVATE mgba_libretro android log)
